  function parseCSV(text) {
    text=text.replace(/^\uFEFF/,'');
    const out=[]; let row=[],field='',quoted=false;
    for(let i=0;i<text.length;i++){
      const c=text[i],n=text[i+1];
      if(quoted){
        if(c==='"'&&n==='"'){field+='"';i++;}
        else if(c==='"') quoted=false;
        else field+=c;
      }else{
        if(c==='"') quoted=true;
        else if(c===','){row.push(field);field='';}
        else if(c==='\n'){row.push(field);out.push(row);row=[];field='';}
        else if(c!=='\r') field+=c;
      }
    }
    if(field.length||row.length){row.push(field);out.push(row);}
    if(!out.length) return [];
    const headers=out[0].map(h=>h.trim());
    return out.slice(1).filter(r=>r.some(x=>String(x).trim()!=='')).map(r=>{
      const obj={}; headers.forEach((h,i)=>obj[h]=r[i]??''); return obj;
    });
  }

  function parseClock(clock) {
    const m=String(clock||'').trim().match(/^(\d{1,2}):(\d{2})\s*(am|pm)$/i);
    if(!m) return null;
    let h=+m[1],min=+m[2],ap=m[3].toLowerCase();
    if(ap==='pm'&&h<12) h+=12;
    if(ap==='am'&&h===12) h=0;
    return {h,min};
  }

  function parseMessageDate(value) {
    const s=String(value||'').trim(); if(!s) return null;
    const now=new Date(); let m=s.match(/^Today\s+(.+)$/i);
    if(m){const c=parseClock(m[1]);if(!c)return null;return new Date(now.getFullYear(),now.getMonth(),now.getDate(),c.h,c.min).toISOString();}
    m=s.match(/^Yesterday\s+(.+)$/i);
    if(m){const c=parseClock(m[1]);if(!c)return null;return new Date(now.getFullYear(),now.getMonth(),now.getDate()-1,c.h,c.min).toISOString();}
    const d=new Date(s.replace(/^[A-Za-z]{3}\s+/,''));
    return Number.isNaN(+d)?null:d.toISOString();
  }

    for(const file of [...files]){
      if(!file.name.toLowerCase().endsWith('.csv')) continue;
      try{parseCSV(await file.text()).forEach(r=>{const row=cloudRow(r,file.name);if(Number.isFinite(row.message_id))all.push(row);});}
      catch(e){console.error('CSV parse failed',file.name,e);}
    }
    if(all.length) await compareRows(all);
  }

  async function unlock() {
    const entered=pwInput.value;
    if(!entered){errorEl.textContent='Enter the access key.';return;}
    unlockBtn.disabled=true;unlockBtn.textContent='Unlocking…';errorEl.textContent='';
    try{
      password=entered;
      const result=await request('PING');
      if(!result.storageConfigured){
        throw new Error('Password accepted, but Supabase storage is not configured on the deployment.');
      }
      unlocked=true;
      sessionStorage.setItem(SESSION_KEY,entered);
      lock.classList.add('hidden');
      setStatus(`Cloud connected${result.keySource?` · ${result.keySource}`:''}`,'ok');
      await loadCloud();
    }catch(e){
      unlocked=false;password='';sessionStorage.removeItem(SESSION_KEY);
      errorEl.textContent=e.message||'Invalid key or cloud connection unavailable.';setStatus('Cloud locked');console.error(e);
    }finally{unlockBtn.disabled=false;unlockBtn.textContent='Unlock GeekMail';}
  }

  async function trySaved() {
    if(!password) return;
    pwInput.value=password;
    try{
      const result=await request('PING');
      if(!result.storageConfigured) throw new Error('Storage unavailable');
      unlocked=true;lock.classList.add('hidden');setStatus(`Cloud connected${result.keySource?` · ${result.keySource}`:''}`,'ok');await loadCloud();
    }catch{sessionStorage.removeItem(SESSION_KEY);password='';unlocked=false;pwInput.value='';lock.classList.remove('hidden');setStatus('Cloud locked');}
  }

  unlockBtn.addEventListener('click',unlock);
  pwInput.addEventListener('keydown',e=>{if(e.key==='Enter')unlock();});
  loadBtn.addEventListener('click',loadCloud);
  manageBtn.addEventListener('click',openManage);
  syncBtn.addEventListener('click',()=>document.querySelector('#fileInput')?.click());
  document.querySelector('#gmManageRefresh').addEventListener('click',openManage);
  document.querySelector('#gmNewRecord').addEventListener('click',()=>openEditor());
  document.querySelector('#gmManageSearch').addEventListener('input',renderManage);
  document.querySelector('#gmSaveRecord').addEventListener('click',saveEditor);
  document.querySelector('#gmMergeSync').addEventListener('click',()=>syncRows('merge'));
  document.querySelector('#gmReplaceSync').addEventListener('click',()=>syncRows('replace'));

  manageModal.addEventListener('click',e=>{
    const edit=e.target.closest('[data-edit]');if(edit){const row=dbRows.find(r=>Number(r.message_id)===Number(edit.dataset.edit));if(row)openEditor(row);return;}
    const del=e.target.closest('[data-delete]');if(del){deleteRecord(del.dataset.delete);return;}
  });
  document.addEventListener('click',e=>{
    const close=e.target.closest('[data-close]');if(!close)return;
    ({manage:manageModal,editor:editorModal,sync:syncModal}[close.dataset.close])?.classList.add('hidden');
  });

  const fileInput=document.querySelector('#fileInput');
  fileInput?.addEventListener('change',e=>{if(!loadingFromCloud&&e.target.files?.length)importFiles(e.target.files);},{capture:true});
  const drop=document.querySelector('#drop');
  drop?.addEventListener('drop',e=>{if(!loadingFromCloud&&e.dataTransfer?.files?.length)importFiles(e.dataTransfer.files);});

  trySaved();
})();
