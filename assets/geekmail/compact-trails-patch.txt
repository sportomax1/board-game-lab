  function clean(v){ return String(v ?? '').replace(/\s+/g,' ').trim(); }
  function norm(v){ return clean(v).toLowerCase(); }
  function esc(v){ return String(v ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
  function ts(row){ const n=Date.parse(row.message_ts || row.message_date || ''); return Number.isFinite(n) ? n : 0; }
  function ownUser(){ return clean(document.querySelector('#myUser')?.value || 'sportomax'); }
  function isSent(row){ return norm(row.from_username) === norm(ownUser()); }
  function otherUser(row){ return clean(isSent(row) ? row.to_username : row.from_username) || '(unknown)'; }
  function shortDate(row){
    const n=ts(row); if(!n) return clean(row.message_date).slice(0,10);
    const d=new Date(n); return d.toLocaleDateString(undefined,{month:'2-digit',day:'2-digit',year:'2-digit'});
  }
  function fullDate(row){
    const n=ts(row); if(!n) return clean(row.message_date);
    return new Date(n).toLocaleString(undefined,{year:'numeric',month:'short',day:'numeric',hour:'numeric',minute:'2-digit'});
  }
  function exactText(text){
    return [...document.querySelectorAll('button,a,[role="tab"],span,div')].find(el => clean(el.textContent).toLowerCase()===text.toLowerCase() && el.children.length<=3) || null;
  }
  function closestMatching(start, test, max=6){
    let el=start;
    for(let i=0; el && i<max; i++,el=el.parentElement){ if(test(el)) return el; }
    return null;
  }

  function createModals(){

    document.body.appendChild(controlsModal);
    controlsModal.querySelector('.gm-overlay-x').onclick=()=>controlsModal.classList.add('hidden');

    detailModal.className='gm-overlay hidden';
    detailModal.innerHTML=`<div class="gm-overlay-card"><div class="gm-overlay-head"><div><strong id="gmDetailTitle">Message</strong><br><span id="gmDetailMeta"></span></div><button class="gm-overlay-x" type="button">✕</button></div><div class="gm-overlay-body"><div class="gm-detail-grid" id="gmDetailGrid"></div><div class="gm-detail-body" id="gmDetailBody"></div></div></div>`;
    document.body.appendChild(detailModal);
    detailModal.querySelector('.gm-overlay-x').onclick=()=>detailModal.classList.add('hidden');
    detailModal.addEventListener('click',e=>{if(e.target===detailModal)detailModal.classList.add('hidden');});
  }

  function moveSection(el,label){
    if(!el || !controlsModal || el.closest('#gmControlsStack')) return;
    const section=document.createElement('section'); section.className='gm-controls-section';
    const title=document.createElement('div'); title.className='gm-controls-label'; title.textContent=label;
    section.append(title,el); controlsModal.querySelector('#gmControlsStack').appendChild(section);
  }

  function compactShell(){
    if(document.querySelector('.gm-data-bar')) return;
    createModals();

    const heading=[...document.querySelectorAll('h1,h2,h3,strong')].find(el=>/^GeekMail CSV Explorer$/i.test(clean(el.textContent)));
    const addCsv=exactText('Add CSV');
    const actionRow=closestMatching(addCsv,el=>{const t=clean(el.innerText);return /Add CSV/i.test(t)&&/Open BGG GeekMail/i.test(t)&&/Columns/i.test(t)&&/Clear/i.test(t);},7);
    const searchInput=[...document.querySelectorAll('input')].find(el=>/subject, username, ID, body, folder/i.test(el.getAttribute('placeholder')||''));
    const filterPanel=closestMatching(searchInput,el=>{const t=clean(el.innerText);return /My BGG username/i.test(t)&&/Rows per page/i.test(t)&&/Folder/i.test(t)&&/Person/i.test(t);},6);
    const statCandidates=[...document.querySelectorAll('div,section')].filter(el=>{const t=clean(el.innerText);return /messages/i.test(t)&&/contacts/i.test(t)&&/longest wait for reply/i.test(t);}).sort((a,b)=>clean(a.innerText).length-clean(b.innerText).length);
    const stats=statCandidates[0]||null;

    let headerWrap=null;
    if(heading){
      const p=heading.parentElement;
      if(p && !/Add CSV/i.test(clean(p.innerText))) p.classList.add('gm-shell-hidden');
      else heading.classList.add('gm-shell-hidden');
      headerWrap=p?.parentElement||null;
    }
    if(headerWrap && actionRow && headerWrap.contains(actionRow)) headerWrap.classList.add('gm-shell-collapsed');

    moveSection(actionRow,'Actions & database');
    moveSection(filterPanel,'Search & filters');
    moveSection(stats,'Statistics');

    [actionRow,filterPanel,stats].forEach(el=>{
      const old=el?.parentElement;
      if(old && old!==controlsModal && old.children.length===0) old.classList.add('gm-shell-hidden');
    });

    dataBar=document.createElement('div'); dataBar.className='gm-data-bar';
    dataBar.innerHTML=`<span class="gm-data-brand">✉ GeekMail</span><span class="gm-data-summary" id="gmDataSummary">Data-first view</span><span class="gm-data-spacer"></span><button class="gm-shell-btn" id="gmControlsBtn" type="button">☰ Controls</button><button class="gm-shell-btn" id="gmReloadCompact" type="button">↻ Data</button>`;
    if(tabRow?.parentElement) tabRow.parentElement.insertBefore(dataBar,tabRow); else document.body.prepend(dataBar);
    dataBar.querySelector('#gmControlsBtn').onclick=()=>controlsModal.classList.remove('hidden');
    dataBar.querySelector('#gmReloadCompact').onclick=()=>loadRows(true);

    const summaryBits=[];
    const bodyText=clean(document.body.innerText);
    const m=bodyText.match(/([\d,]+)\s+messages/i); if(m) summaryBits.push(`${m[1]} msgs`);
    const c=bodyText.match(/([\d,]+)\s+contacts/i); if(c) summaryBits.push(`${c[1]} contacts`);
    if(summaryBits.length) dataBar.querySelector('#gmDataSummary').textContent=summaryBits.join(' · ');
  }

  function createCompactPanel(){
    compactPanel=document.createElement('section'); compactPanel.className='gm-compact-panel';
    compactPanel.innerHTML=`
      <div class="gm-compact-toolbar">
        <input class="gm-compact-search" id="gmCompactSearch" placeholder="Quick search: contact or subject…">
        <div class="gm-seg"><button type="button" id="gmGrouped" class="active">Grouped</button><button type="button" id="gmFlat">Flat</button></div>
        <select class="gm-compact-select" id="gmContactSort" title="Contact sort"><option value="recent">Recent contacts</option><option value="alpha">A–Z contacts</option><option value="count">Most messages</option></select>
        <select class="gm-compact-select" id="gmMessageSort" title="Message order"><option value="newest">Newest first</option><option value="oldest">Oldest first</option></select>
        <select class="gm-compact-select" id="gmFolderFilter" title="Folder"><option value="all">All folders</option></select>
        <span class="gm-compact-count" id="gmCompactCount">Load Compact to view data</span>
      </div>
      <div class="gm-compact-body" id="gmCompactBody"><div class="gm-loading">Compact view will load the full GeekMail archive.</div></div>`;
    const parent=tabRow?.parentElement||document.body; parent.insertBefore(compactPanel,tabRow?.nextSibling||null);

    compactPanel.querySelector('#gmCompactSearch').addEventListener('input',e=>{searchText=e.target.value;render();});
    compactPanel.querySelector('#gmGrouped').onclick=()=>{mode='grouped';setModeButtons();render();};
    compactPanel.querySelector('#gmFlat').onclick=()=>{mode='flat';setModeButtons();render();};
    compactPanel.querySelector('#gmContactSort').onchange=e=>{contactSort=e.target.value;render();};
    compactPanel.querySelector('#gmMessageSort').onchange=e=>{messageSort=e.target.value;render();};
    compactPanel.querySelector('#gmFolderFilter').onchange=e=>{folderFilter=e.target.value;render();};
  }

  function setModeButtons(){
    compactPanel?.querySelector('#gmGrouped')?.classList.toggle('active',mode==='grouped');
    compactPanel?.querySelector('#gmFlat')?.classList.toggle('active',mode==='flat');
  }

  function setupTabs(){
    const contact=exactText('Contact trails');
    const data=exactText('Data table');
    const getNew=exactText('Get New Data');
    if(!contact || !data) return false;
    tabRow=contact.parentElement;
    viewHost=tabRow?.parentElement||null;

    compactTab=document.createElement('button');
    compactTab.type='button'; compactTab.textContent='Compact'; compactTab.className=`${contact.className||''} gm-compact-tab`.trim();
    compactTab.removeAttribute('id'); compactTab.setAttribute('aria-selected','false');
    contact.insertAdjacentElement('afterend',compactTab);
    compactTab.addEventListener('click',showCompact);
    [contact,data,getNew].filter(Boolean).forEach(tab=>tab.addEventListener('click',hideCompact));

    createCompactPanel();
    compactShell();
    return true;
  }

  function markOriginalViewNodes(){
    if(!viewHost || !tabRow) return;
    [...viewHost.children].forEach(child=>{
      if(child===tabRow||child===compactPanel||child===dataBar||child===controlsModal||child===detailModal) return;
      child.classList.add('gm-original-view-node');
      child.classList.toggle('gm-hide-for-compact',compactActive);
    });
  }

  async function showCompact(e){
    e?.preventDefault(); e?.stopPropagation();
    compactActive=true; document.body.classList.add('gm-compact-active'); compactTab?.classList.add('gm-tab-active'); compactTab?.setAttribute('aria-selected','true');
    markOriginalViewNodes();
    if(!loaded) await loadRows(false); else render();
  }
  function hideCompact(){
    compactActive=false; document.body.classList.remove('gm-compact-active'); compactTab?.classList.remove('gm-tab-active'); compactTab?.setAttribute('aria-selected','false');
    document.querySelectorAll('.gm-original-view-node').forEach(el=>el.classList.remove('gm-hide-for-compact'));
  }

  async function loadRows(force=false){
    if(loading || (loaded&&!force)) return;
    loading=true;
    const body=compactPanel?.querySelector('#gmCompactBody'); if(body) body.innerHTML='<div class="gm-loading">Loading full GeekMail archive…</div>';
    try{
      const password=sessionStorage.getItem(SESSION_KEY)||'';
      const response=await fetch(API,{method:'POST',headers:{'Content-Type':'application/json'},cache:'no-store',body:JSON.stringify({action:'READ',password,limit:10000})});
      const data=await response.json().catch(()=>({}));
      if(!response.ok||!data.ok) throw new Error(data.error||`GeekMail read failed (${response.status})`);
      rows=Array.isArray(data.rows)?data.rows:[]; loaded=true;
      populateFolders();
      const contacts=new Set(rows.map(otherUser).filter(Boolean));
      const summary=dataBar?.querySelector('#gmDataSummary'); if(summary) summary.textContent=`${rows.length.toLocaleString()} msgs · ${contacts.size.toLocaleString()} contacts`;
      render();
    }catch(err){ if(body) body.innerHTML=`<div class="gm-empty">Compact view could not load: ${esc(err.message)}</div>`; }
    finally{loading=false;}
  }

  function populateFolders(){
    const sel=compactPanel?.querySelector('#gmFolderFilter'); if(!sel) return;
    const old=sel.value||'all';
    const folders=[...new Set(rows.map(r=>clean(r.folder)).filter(Boolean))].sort((a,b)=>a.localeCompare(b));
    sel.innerHTML='<option value="all">All folders</option>'+folders.map(f=>`<option value="${esc(f)}">${esc(f)}</option>`).join('');
    if([...sel.options].some(o=>o.value===old)) sel.value=old;
  }

  function filteredRows(){
    const q=norm(searchText);
    return rows.filter(r=>{
      if(folderFilter!=='all' && clean(r.folder)!==folderFilter) return false;
      if(!q) return true;
      return [otherUser(r),r.subject,r.message_date,r.folder].some(v=>norm(v).includes(q));
    });
  }

  function sortedMessages(list){ return [...list].sort((a,b)=>messageSort==='oldest'?ts(a)-ts(b):ts(b)-ts(a)); }

  function render(){
    if(!compactPanel||!loaded) return;
    const body=compactPanel.querySelector('#gmCompactBody'); const list=filteredRows();
    const contacts=new Map();
    list.forEach(r=>{const u=otherUser(r); if(!contacts.has(u))contacts.set(u,[]); contacts.get(u).push(r);});
    compactPanel.querySelector('#gmCompactCount').textContent=`${list.length.toLocaleString()} messages · ${contacts.size.toLocaleString()} contacts`;
    if(!list.length){body.innerHTML='<div class="gm-empty">No messages match the compact filters.</div>';return;}
    if(mode==='flat') renderFlat(body,list); else renderGrouped(body,contacts);
  }

  function renderGrouped(body,contacts){
    let groups=[...contacts.entries()].map(([user,msgs])=>({user,msgs:sortedMessages(msgs),recent:Math.max(...msgs.map(ts),0)}));
    groups.sort((a,b)=>contactSort==='alpha'?a.user.localeCompare(b.user):contactSort==='count'?b.msgs.length-a.msgs.length:b.recent-a.recent);
    body.innerHTML=`<div class="gm-contact-grid">${groups.map(g=>`<section class="gm-contact-box"><div class="gm-contact-box-head"><span class="gm-contact-name" title="${esc(g.user)}">${esc(g.user)}</span><span class="gm-contact-meta">${g.msgs.length} msg${g.msgs.length===1?'':'s'}</span></div><div>${g.msgs.map(messageRowHtml).join('')}</div></section>`).join('')}</div>`;
    bindRows(body);
  }

  function messageRowHtml(r){
    const sent=isSent(r), dir=sent?'→':'←';
    return `<div class="gm-mini-row" data-mid="${esc(r.message_id)}" title="${esc(fullDate(r))} · ${sent?'Sent':'Received'}"><span class="gm-mini-date">${esc(shortDate(r))}</span><span class="gm-mini-dir ${sent?'sent':'received'}">${dir}</span><span class="gm-mini-subject" title="${esc(r.subject||'(no subject)')}">${esc(r.subject||'(no subject)')}</span></div>`;
  }

  function renderFlat(body,list){
    const ordered=sortedMessages(list);
    body.innerHTML=`<div class="gm-flat-wrap"><div class="gm-flat-head"><span>Date</span><span></span><span>Contact</span><span>Subject</span></div>${ordered.map(r=>{const sent=isSent(r);return `<div class="gm-flat-row" data-mid="${esc(r.message_id)}"><span class="gm-mini-date">${esc(shortDate(r))}</span><span class="gm-mini-dir ${sent?'sent':'received'}">${sent?'→':'←'}</span><span class="gm-flat-user" title="${esc(otherUser(r))}">${esc(otherUser(r))}</span><span class="gm-flat-subject" title="${esc(r.subject||'(no subject)')}">${esc(r.subject||'(no subject)')}</span></div>`;}).join('')}</div>`;
    bindRows(body);
  }

  function bindRows(root){ root.querySelectorAll('[data-mid]').forEach(el=>el.addEventListener('click',()=>openDetail(el.dataset.mid))); }
  function openDetail(id){
    const r=rows.find(x=>String(x.message_id)===String(id)); if(!r||!detailModal)return;
    const sent=isSent(r); detailModal.querySelector('#gmDetailTitle').textContent=r.subject||'(no subject)';
    detailModal.querySelector('#gmDetailMeta').textContent=`${sent?'Sent to':'Received from'} ${otherUser(r)} · ${fullDate(r)}`;
    const items=[['Contact',otherUser(r)],['Direction',sent?'Sent →':'← Received'],['Date',fullDate(r)],['Folder',r.folder||''],['Message ID',r.message_id||''],['Route',`${r.from_username||''} → ${r.to_username||''}`]];
    detailModal.querySelector('#gmDetailGrid').innerHTML=items.map(([k,v])=>`<b>${esc(k)}</b><span>${esc(v)}</span>`).join('');
    detailModal.querySelector('#gmDetailBody').textContent=r.body_text||r.body_html||'(No message body)'; detailModal.classList.remove('hidden');
  }

  function boot(){
    let attempts=0;
    const timer=setInterval(()=>{
      attempts++;
      if(setupTabs()||attempts>80) clearInterval(timer);
    },100);
    const observer=new MutationObserver(()=>{if(compactActive)markOriginalViewNodes();});
    observer.observe(document.body,{childList:true,subtree:false});
  }

  boot();
})();
