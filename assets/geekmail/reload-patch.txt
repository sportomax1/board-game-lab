<script>
(() => {
  if (window.__GEEKMAIL_SYNC_RELOAD_PATCH__) return;
  window.__GEEKMAIL_SYNC_RELOAD_PATCH__ = true;

  const deltaContent = document.querySelector('#gmDeltaContent');
  const syncModal = deltaContent?.closest('.gm-modal');
  const status = document.querySelector('.gm-cloud-pill');
  const loadBtn = [...document.querySelectorAll('button')].find(btn => btn.textContent.includes('Load DB'));
  if (!syncModal || !status || !loadBtn) return;

  let wasOpen = !syncModal.classList.contains('hidden');
  const observer = new MutationObserver(() => {
    const isOpen = !syncModal.classList.contains('hidden');
    if (wasOpen && !isOpen && /^(Merged|Replaced)\b/.test(status.textContent || '')) {
      let attempts = 0;
      const refreshWhenReady = () => {
        attempts += 1;
        if (!loadBtn.disabled) {
          loadBtn.click();
          return;
        }
        if (attempts < 20) setTimeout(refreshWhenReady, 50);
      };
      setTimeout(refreshWhenReady, 0);
    }
    wasOpen = isOpen;
  });

  observer.observe(syncModal, { attributes: true, attributeFilter: ['class'] });
})();
</script>
